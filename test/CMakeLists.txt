cmake_minimum_required(VERSION 3.16)
project(tv4_bench LANGUAGES C CXX)
enable_testing()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(MSVC)
    add_compile_options(/utf-8)
endif()
set(ROOT "${CMAKE_CURRENT_LIST_DIR}/..")
set(CORE "${ROOT}/lib/TV4Core/src")
add_library(core "${CORE}/Calibration.cpp" "${CORE}/DemoService.cpp"
                 "${CORE}/RPMService.cpp" "${CORE}/StatisticsService.cpp")
target_include_directories(core PUBLIC "${CORE}")
add_executable(test_core test_core.cpp)
target_link_libraries(test_core PRIVATE core)
add_test(NAME core COMMAND test_core)
add_executable(test_storage test_storage.cpp "${ROOT}/src/storage/SettingsService.cpp")
target_include_directories(test_storage PRIVATE support "${ROOT}/src/storage")
target_link_libraries(test_storage PRIVATE core)
add_test(NAME storage COMMAND test_storage)

set(LVGL_DIR "" CACHE PATH "Path to LVGL 8.3.11 sources, for the display bench")
if(LVGL_DIR)
    file(GLOB_RECURSE LVGL_C CONFIGURE_DEPENDS "${LVGL_DIR}/src/*.c")
    add_library(lvgl STATIC ${LVGL_C})
    target_compile_features(lvgl PRIVATE c_std_11)
    target_include_directories(lvgl PUBLIC "${LVGL_DIR}" "${ROOT}/include")
    target_compile_definitions(lvgl PUBLIC LV_CONF_INCLUDE_SIMPLE _CRT_SECURE_NO_WARNINGS)
    add_executable(test_ui test_ui.cpp "${ROOT}/lib/PanelUI/src/UIManager.cpp"
                            "${ROOT}/lib/PanelUI/src/tv4_text_15.c")
    target_compile_features(test_ui PRIVATE c_std_11)
    target_include_directories(test_ui PRIVATE "${ROOT}/lib/PanelUI/src")
    target_link_libraries(test_ui PRIVATE lvgl core)
    add_test(NAME ui COMMAND test_ui)
endif()
